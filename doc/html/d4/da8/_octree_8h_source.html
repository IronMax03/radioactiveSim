101 inline void eval()
102 {
103 // recursion base case.
104 if (is_leaf())
105 {
106 if (particle != nullptr)
107 {
108 center_of_charge = particle->position;
109 total_charge = particle->electric_charge;
110 }
111
112 return;
113 }
114
115 center_of_charge = vector3<T>{0, 0, 0};
116 total_charge = 0;
117
118 for (std::unique_ptr<lazy_node<T>>& c: childs)
119 {
120
121 if (c != nullptr)
122 {
123 if (is_leaf() && c->particle != nullptr)
124 continue; // if the current node is a leaf and the child has a particle, we can skip the evaluation of the child since we will use the particle of the child to calculate the center of charge and total charge of the current node.;
125
126 c->eval();
127 center_of_charge = (total_charge*center_of_charge + c->total_charge*c->center_of_charge)/(total_charge + c->total_charge);
128 total_charge += c->total_charge;
129 }
130 }
