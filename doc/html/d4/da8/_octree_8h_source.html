46 constexpr size_t child_index(const vector3<T>& p) const
47 {
48 if (p.x < bounding_box.center.x - bounding_box.length/2 ||
49 p.y < bounding_box.center.y - bounding_box.length/2 ||
50 p.z < bounding_box.center.z - bounding_box.length/2 ||
51 p.x > bounding_box.center.x + bounding_box.length/2 ||
52 p.y > bounding_box.center.y + bounding_box.length/2 ||
53 p.z > bounding_box.center.z + bounding_box.length/2)
54 throw std::runtime_error("lazy_node.child_index(p): Particle position must be within the bounding box of the node.");
55
56 size_t child_index = 0;
57
58 // Determine the child index of the particle based on morton indices.
59 if (p.x >= bounding_box.center.x)
60 child_index = child_index | 0b001; // equivalent to child_index += 1;
61 if (p.y >= bounding_box.center.y)
62 child_index = child_index | 0b010;// equivalent to child_index += 2;
63 if (p.z >= bounding_box.center.z)
64 child_index = child_index | 0b100;// equivalent to child_index += 4;
65
66 return child_index;

101 inline void eval()
102 {
103 // recursion base case.
104 if (is_leaf())
105 {
106 if (particle != nullptr)
107 {
108 center_of_charge = particle->position;
109 total_charge = particle->electric_charge;
110 }
111
112 return;
113 }
114
115 center_of_charge = vector3<T>{0, 0, 0};
116 total_charge = 0;
117
118 for (std::unique_ptr<lazy_node<T>>& c: childs)
119 {
120
121 if (c != nullptr)
122 {
123 if (is_leaf() && c->particle != nullptr)
124 continue; // if the current node is a leaf and the child has a particle, we can skip the evaluation of the child since we will use the particle of the child to calculate the center of charge and total charge of the current node.;
125
126 c->eval();
127 int combined = total_charge + c->total_charge;
128 if (combined != 0)
129 center_of_charge = (total_charge*center_of_charge + c->total_charge*c->center_of_charge)/combined;
130 total_charge = combined;
131 }
132 }
133 }
